DECLARE
   v_net NUMBER := 123.45; -- ΚΑΘΑΡΗ ΤΙΜΗ
   v_rate NUMBER := 0.24; -- ΦΠΑ 24%
   v_tax NUMBER;
   v_gross NUMBER;
   v_pow NUMBER;
  v_mod NUMBER;
BEGIN 
 v_tax := v_net * v_rate; --  πολλαπλασιασμος 
 v_gross := v_net + v_tax; -- αθροισμα 
 v_pow := 2 ** 3; -- υψωση στο τετραγωνο
 v_mod := MOD(17,5); -- υπολοιπο διαιρεσης 
 DBMS_OUTPUT.PUT_LINE('TAX=' || ROUND(v_tax,2)||'gross='||ROUND (v_gross,2));
 DBMS_OUTPUT.PUT_LINE('2**3='||v_pow||' mod(17,5)'=||v_mod);
END;
/
