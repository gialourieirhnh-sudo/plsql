CREATE OR REPLACE PROCEDURE pl(v1 NUMBER , v2 NUMBER) AS
BEGIN 
  DBMS_OUTPUT.PUT_LINE(v1 || '+' || v2 || ' = ' || (v1+v2));
  DBMS_OUTPUT.PUT_LINE(v1 || '-' || v2 || ' = ' || (v1-v2));
  DBMS_OUTPUT.PUT_LINE(v1 || '*' || v2 || ' = ' || (v1*v2));
  DBMS_OUTPUT.PUT_LINE(v1 || '/' || v2 || ' = ' || (v1/v2));
  
END;
/
BEGIN
  pl(5,4);
  pl(6,2);
  pl(8, 3);
END;
/
